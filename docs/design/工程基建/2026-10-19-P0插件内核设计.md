## 2. Context 契约详细设计

### 2.1 核心接口

```typescript
// packages/plugin-core/src/types.ts

/**
 * 服务的唯一标识符。
 * 用字符串而不是 Symbol，以便跨 bundle 访问、日志、JSON 序列化。
 */
export type ServiceKey<T = any> = string & { readonly __nominal: T }

/**
 * 创建类型安全的服务 key。
 * 
 * 使用方式：
 *   const TOOLS = createServiceKey<ToolRegistry>('tools')
 *   ctx.provide(TOOLS, toolRegistry)
 *   ctx.require(TOOLS)  // 类型自动推导为 ToolRegistry
 */
export function createServiceKey<T>(name: string): ServiceKey<T> {
  return name as ServiceKey<T>
}

/**
 * 日志接口。带 pluginId 前缀。
 */
export interface PluginLogger {
  debug(message: string, data?: any): void
  info(message: string, data?: any): void
  warn(message: string, data?: any): void
  error(message: string, error?: any): void
}

/**
 * 核心插件上下文接口。
 * 
 * 不变量：
 * - 所有贡献都通过这个对象完成；插件不能直接 import 实现
 * - 注册返回 disposer；PluginHost 持有作用域
 * - effect 与事件监听都自动卸载
 */
export interface PluginContext {
  /**
   * 插件的全局唯一 ID。必须符合 `com.example.plugin-name` 格式。
   */
  readonly pluginId: string

  /**
   * 提供（或装配）一个服务。
   * 
   * @param key 服务标识符
   * @param value 服务实例或工厂函数
   * @returns 解析后的服务实例
   * 
   * 规则：
   * - 同一 key 只能 provide 一次；重复 provide 抛 `DuplicateServiceError`
   * - 如果 value 是函数，首次 require 时调用（lazy）
   * - 随插件卸载自动移除；依赖此服务的插件不会重新激活
   */
  provide<T>(key: ServiceKey<T>, value: T | (() => T)): T

  /**
   * 获取一个服务。
   * 
   * @returns 服务实例，或 undefined（不存在或被卸载）
   * 
   * 规则：
   * - 不抛错；缺失返回 undefined
   * - 用于可选依赖
   */
  get<T>(key: ServiceKey<T>): T | undefined

  /**
   * 强制获取一个服务。
   * 
   * @throws `MissingServiceError` 如果不存在
   * 
   * 规则：
   * - 用于必需依赖
   * - 激活期调用 require 失败直接标记插件为 failed
   */
  require<T>(key: ServiceKey<T>): T

  /**
   * 注册一个可逆副作用（清理函数）。
   * 
   * @param dispose 卸载时调用的函数，支持异步
   * 
   * 规则：
   * - 按 LIFO 顺序执行（后注册先清理）
   * - 异步清理严格串行，一个接一个 await
   * - 清理中失败记录但继续；最后用 AggregateError 汇总抛出
   * - 卸载完毕后若再 effect()，抛 `ScopeAlreadyDisposedError`
   * 
   * 示例：
   *   ctx.effect(() => {
   *     const timer = setInterval(() => { ... }, 1000)
   *     return () => clearInterval(timer)
   *   })
   */
  effect(dispose: () => void | Promise<void>): void

  /**
   * 订阅事件。
   * 
   * @param name 事件名（来自 LumiiEvents）
   * @param listener 监听函数
   * 
   * 规则：
   * - 监听器随插件卸载自动移除
   * - 同一事件可注册多个监听器
   * - 监听器异常处理由事件模式决定（见 §2.3）
   */
  on<K extends keyof LumiiEvents>(
    name: K,
    listener: (payload: LumiiEvents[K]) => void | Promise<void>,
  ): void

  /**
   * 同步发射事件（fire-and-forget）。
   * 
   * @param name 事件名
   * @param payload 事件数据
   * 
   * 规则：
   * - 按注册顺序调用所有监听器（不等待异步）
   * - 某个监听器异常不影响后续监听器；错误被记录
   * - 不返回值
   * 
   * 场景：用于通知性事件（"插件已激活"、"工具已执行"）
   */
  emit<K extends keyof LumiiEvents>(name: K, payload: LumiiEvents[K]): void

  /**
   * 瀑布流模式事件（环绕中间件）。
   * 
   * @param name 事件名
   * @param payload 事件数据
   * @returns 最后一个监听器返回的（可能被修改的）payload
   * 
   * 规则：
   * - 每个监听器收到 (payload, next) 两个参数
   * - 调用 next() 继续执行下一个监听器；不调用则短路
   * - 可修改 payload 并返回，后续监听器收到新值
   * - 如果某个监听器抛错，错误向上传播，后续监听器不执行
   * 
   * 场景：权限检查、数据预处理、工具执行前校验
   * 
   * 示例：
   *   ctx.waterfall('tool:pre-execute', {
   *     toolName: 'search',
   *     params: { query: '...' },
   *   }, async (payload, next) => {
   *     if (policy.canExecute(payload.toolName)) {
   *       return await next()  // -> 下一个监听器的返回值
   *     }
   *     throw new PermissionError(...)
   *   })
   */
  waterfall<K extends keyof LumiiEvents>(
    name: K,
    payload: LumiiEvents[K],
  ): Promise<LumiiEvents[K]>

  /**
   * 串行模式事件（按顺序 await）。
   * 
   * @returns 最后一个监听器的返回值
   * 
   * 规则：
   * - 按注册顺序逐个 await 监听器
   * - 如果某个监听器抛错，错误向上传播，后续不执行
   * - 每个监听器都可修改状态
   * 
   * 场景：协调多个插件（"关闭前都同步数据"）
   */
  serial<K extends keyof LumiiEvents>(
    name: K,
    payload: LumiiEvents[K],
  ): Promise<LumiiEvents[K]>

  /**
   * 并行模式事件（全部并发 await，无返回值）。
   * 
   * 规则：
   * - 所有监听器并发执行，等待全部完成
   * - 某个监听器异常记录但不影响其他监听器
   * - 都异常时抛 AggregateError
   * 
   * 场景：独立的通知（多个插件各自做清理）
   */
  parallel<K extends keyof LumiiEvents>(
    name: K,
    payload: LumiiEvents[K],
  ): Promise<void>

  /**
   * 日志（带 pluginId 前缀）。
   */
  readonly log: PluginLogger
}
```

### 2.2 Context 的实现细节

**创建 Context 的工厂函数**：

```typescript
// packages/plugin-core/src/context.ts

export function createPluginContext(
  pluginId: string,
  eventBus: EventBus,
  scope: PluginScope,
): PluginContext {
  const services = new Map<ServiceKey, any>()
  const serviceFactories = new Map<ServiceKey, () => any>()

  return {
    pluginId,

    provide<T>(key: ServiceKey<T>, value: T | (() => T)): T {
      if (services.has(key) || serviceFactories.has(key)) {
        throw new DuplicateServiceError(this.pluginId, String(key))
      }

      if (typeof value === 'function') {
        serviceFactories.set(key, value as () => T)
        // 不立即调用；延迟到 require 时
        return undefined as any // 占位
      } else {
        services.set(key, value)
        // 注册清理函数：卸载时移除服务，不影响已获取的引用
        scope.effect(() => {
          services.delete(key)
          serviceFactories.delete(key)
        })
        return value
      }
    },

    get<T>(key: ServiceKey<T>): T | undefined {
      if (services.has(key)) return services.get(key)
      if (serviceFactories.has(key)) {
        const factory = serviceFactories.get(key)!
        const instance = factory()
        services.set(key, instance)
        serviceFactories.delete(key) // 只调一次
        return instance
      }
      return undefined
    },

    require<T>(key: ServiceKey<T>): T {
      const svc = this.get(key)
      if (svc === undefined) {
        throw new MissingServiceError(this.pluginId, String(key))
      }
      return svc
    },

    effect(dispose: () => void | Promise<void>): void {
      scope.effect(dispose)
    },

    on<K extends keyof LumiiEvents>(
      name: K,
      listener: (payload: LumiiEvents[K]) => void | Promise<void>,
    ): void {
      eventBus.on(name, listener)
      scope.effect(() => {
        eventBus.off(name, listener)
      })
    },

    emit<K extends keyof LumiiEvents>(
      name: K,
      payload: LumiiEvents[K],
    ): void {
      eventBus.emit(name, payload)
    },

    waterfall<K extends keyof LumiiEvents>(
      name: K,
      payload: LumiiEvents[K],
    ): Promise<LumiiEvents[K]> {
      return eventBus.waterfall(name, payload)
    },

    serial<K extends keyof LumiiEvents>(
      name: K,
      payload: LumiiEvents[K],
    ): Promise<LumiiEvents[K]> {
      return eventBus.serial(name, payload)
    },

    parallel<K extends keyof LumiiEvents>(
      name: K,
      payload: LumiiEvents[K],
    ): Promise<void> {
      return eventBus.parallel(name, payload)
    },

    log: new PluginLogger(pluginId),
  }
}
```

## 3. Effect 作用域详细设计

### 3.1 核心实现

```typescript
// packages/plugin-core/src/scope.ts

export interface DisposerResult {
  ok: boolean
  errors?: Array<{ step: number; error: unknown }>
}

/**
 * 插件的作用域与资源管理。
 * 
 * 不变量：
 * - 清理按 LIFO 顺序（后注册先清理）
 * - 异步清理严格串行（不并发）
 * - 清理中的异常不阻止其他清理；最后汇总报告
 * - 多次 dispose() 调用是幂等的
 */
export class PluginScope {
  private disposers: Array<() => void | Promise<void>> = []
  private disposed = false
  private disposePromise: Promise<DisposerResult> | null = null

  /**
   * 注册一个清理函数。
   * 
   * @throws ScopeAlreadyDisposedError 如果已卸载
   */
  effect(dispose: () => void | Promise<void>): void {
    if (this.disposed) {
      throw new ScopeAlreadyDisposedError(
        'Cannot register effect after scope has been disposed',
      )
    }
    this.disposers.push(dispose)
  }

  /**
   * 执行所有清理函数。
   * 
   * 返回结果包含：
   * - ok: 是否全部成功
   * - errors: 失败列表 [{ step: 索引, error: 异常 }, ...]
   * 
   * 规则：
   * - 只执行一次（后续调用返回缓存 promise）
   * - 异步清理按 LIFO 串行等待（index 从 N-1 → 0）
   * - 清理失败记录，不中断
   * - 最后一个清理失败则 ok=false
   * 
   * 示例：
   *   const result = await scope.dispose()
   *   if (!result.ok) {
   *     logger.error('Dispose errors:', result.errors)
   *   }
   */
  async dispose(): Promise<DisposerResult> {
    if (this.disposePromise) {
      return this.disposePromise
    }

    this.disposePromise = this._doDispose()
    return this.disposePromise
  }

  private async _doDispose(): Promise<DisposerResult> {
    if (this.disposed) {
      return { ok: true }  // 幂等
    }

    this.disposed = true
    const errors: Array<{ step: number; error: unknown }> = []

    // LIFO: 从最后往前清理
    for (let i = this.disposers.length - 1; i >= 0; i--) {
      try {
        const disposer = this.disposers[i]
        const result = disposer()

        // 如果返回 Promise，等待
        if (result instanceof Promise) {
          await result
        }
      } catch (err) {
        errors.push({ step: i, error: err })
      }
    }

    this.disposers.length = 0

    return {
      ok: errors.length === 0,
      errors: errors.length > 0 ? errors : undefined,
    }
  }
}
```

### 3.2 关键细节说明

**为什么异步清理必须串行？**

反例（并发清理会导致的问题）：
```typescript
// ❌ 错误：并发
const promises = disposers.map(d => Promise.resolve(d()))
await Promise.all(promises)  // 多个 disposer 并发跑

// 问题：disposer A 需要资源 B，但 disposer B 正并发执行中
// → 资源被提前释放 → A 得到 null 或异常
```

正确做法（串行）：
```typescript
// ✅ 正确：串行
for (let i = disposers.length - 1; i >= 0; i--) {
  await disposers[i]()  // 一个等完再执行下一个
}
// 前提条件：disposers[i-1] 必须在 disposers[i] 启动前完全清理
```

### 4.1 事件类型定义

```typescript
// packages/plugin-core/src/events.ts

/**
 * Lumii 系统的全部事件。
 * 
 * 命名规则：`domain:action`
 * 
 * 事件模式由订阅侧决定（emit vs waterfall vs serial）；
 * 发射侧只负责调 emit()，EventBus 根据事件类型自动选择模式。
 */
export interface LumiiEvents {
  // 应用生命周期
  'app:ready': {
    /** 就绪时间戳（ms） */
    at: number
    /** 主窗口是否显示 */
    mainWindowShown: boolean
  }

  'app:before-quit': {
    /** 是否强制退出（Ctrl+C） */
    forced: boolean
  }

  // 插件生命周期
  'plugin:discovered': {
    pluginId: string
    version: string
  }

  'plugin:activated': {
    pluginId: string
    /** 激活耗时（ms） */
    durationMs: number
    /** 注册的 effect 数 */
    effectCount: number
  }

  'plugin:deactivated': {
    pluginId: string
    /** 停用原因：disabled / error / uninstall / reload */
    reason: 'disabled' | 'error' | 'uninstall' | 'reload'
    /** 清理耗时（ms） */
    cleanupMs?: number
  }

  'plugin:error': {
    pluginId: string
    error: unknown
    /** 错误发生的阶段 */
    stage: 'activate' | 'effect' | 'event-handler'
    /** 是否已恢复（插件转 FAILED 但 App 继续） */
    recovered: boolean
  }

  // Agent 生命周期
  'agent:message-added': {
    conversationId: string
    messageId: string
    role: 'user' | 'assistant' | 'system'
  }

  'agent:pre-step': {
    conversationId: string
    /** 当前回合的消息数 */
    messageCount: number
  }

  'agent:post-step': {
    conversationId: string
    /** 生成耗时（ms） */
    durationMs: number
  }

  // 工具执行
  'tool:pre-execute': {
    toolName: string
    /** 工具参数 */
    params: unknown
  }

  'tool:post-execute': {
    toolName: string
    /** 执行耗时（ms） */
    durationMs: number
    /** 是否成功 */
    success: boolean
  }

  // 声道（可自定义）
  [key: string]: any
}
```

### 4.2 EventBus 实现

```typescript
// packages/plugin-core/src/event-bus.ts

type Listener<T> = (payload: T) => void | Promise<void>
type WaterfallListener<T> = (
  payload: T,
  next: () => Promise<T>,
) => Promise<T>

export class EventBus {
  private listeners = new Map<string, Set<Listener<any>>>()
  private waterfallListeners = new Map<string, Array<WaterfallListener<any>>>()

  /**
   * 订阅事件。
   */
  on<K extends keyof LumiiEvents>(
    name: K,
    listener: Listener<LumiiEvents[K]>,
  ): void {
    const key = String(name)
    if (!this.listeners.has(key)) {
      this.listeners.set(key, new Set())
    }
    this.listeners.get(key)!.add(listener)
  }

  /**
   * 取消订阅。
   */
  off<K extends keyof LumiiEvents>(
    name: K,
    listener: Listener<LumiiEvents[K]>,
  ): void {
    const key = String(name)
    this.listeners.get(key)?.delete(listener)
  }

  /**
   * 发射事件（fire-and-forget）。
   */
  emit<K extends keyof LumiiEvents>(
    name: K,
    payload: LumiiEvents[K],
  ): void {
    const key = String(name)
    const listeners = this.listeners.get(key)
    if (!listeners) return

    for (const listener of listeners) {
      try {
        const result = listener(payload)
        // 异步监听器不等待，但错误被记录
        if (result instanceof Promise) {
          result.catch((err) => {
            console.error(`[EventBus] emit ${key} listener error:`, err)
          })
        }
      } catch (err) {
        console.error(`[EventBus] emit ${key} listener error:`, err)
      }
    }
  }

  /**
   * 瀑布流模式（环绕中间件）。
   * 
   * 每个监听器收到 (payload, next)；调用 next() 继续。
   */
  async waterfall<K extends keyof LumiiEvents>(
    name: K,
    payload: LumiiEvents[K],
  ): Promise<LumiiEvents[K]> {
    const key = String(name)
    const listeners = this.waterfallListeners.get(key) || []

    let result = payload
    let index = 0

    const next = async (): Promise<LumiiEvents[K]> => {
      if (index >= listeners.length) return result
      const listener = listeners[index++]
      return listener(result, next)
    }

    try {
      return await next()
    } catch (err) {
      // 瀑布流异常直接向上传
      throw err
    }
  }

  /**
   * 串行模式（按顺序 await）。
   */
  async serial<K extends keyof LumiiEvents>(
    name: K,
    payload: LumiiEvents[K],
  ): Promise<LumiiEvents[K]> {
    const key = String(name)
    const listeners = this.listeners.get(key)
    if (!listeners) return payload

    let result = payload
    for (const listener of listeners) {
      result = (await listener(result)) ?? payload
    }
    return result
  }

  /**
   * 并行模式（全部并发 await）。
   */
  async parallel<K extends keyof LumiiEvents>(
    name: K,
    payload: LumiiEvents[K],
  ): Promise<void> {
    const key = String(name)
    const listeners = this.listeners.get(key)
    if (!listeners) return

    const promises = Array.from(listeners).map((listener) =>
      Promise.resolve()
        .then(() => listener(payload))
        .catch((err) => {
          console.error(
            `[EventBus] parallel ${key} listener error:`,
            err,
          )
          throw err
        }),
    )

    const results = await Promise.allSettled(promises)
    const errors = results
      .filter((r) => r.status === 'rejected')
      .map((r) => (r as PromiseRejectedResult).reason)

    if (errors.length > 0) {
      throw new AggregateError(
        errors,
        `parallel event ${key} had ${errors.length} error(s)`,
      )
    }
  }
}
```

```typescript
import Ajv from 'ajv'

/**
 * Manifest 的 JSON Schema。
 */
export const MANIFEST_SCHEMA = {
  type: 'object',
  required: ['id', 'name', 'version', 'kind', 'engines'],
  properties: {
    id: {
      type: 'string',
      pattern: '^[a-z0-9]+([.-][a-z0-9]+)*$',
      minLength: 3,
      maxLength: 255,
    },
    name: {
      type: 'string',
      minLength: 1,
      maxLength: 255,
    },
    version: {
      type: 'string',
      pattern: '^\\d+\\.\\d+\\.\\d+',  // semver
    },
    description: { type: ['string', 'null'] },
    icon: { type: ['string', 'null'] },
    kind: { enum: ['code', 'asset'] },
    main: { type: ['string', 'null'] },
    client: {
      type: 'object',
      properties: {
        entry: { type: 'string' },
        styles: { type: 'array', items: { type: 'string' } },
        external: { type: 'array', items: { type: 'string' } },
      },
      required: ['entry'],
    },
    engines: {
      type: 'object',
      properties: {
        lumii: { type: 'string' },
      },
      required: ['lumii'],
    },
    activation: {
      type: 'array',
      items: { type: 'string' },
    },
    permissions: {
      type: 'array',
      items: {
        enum: [
          'fs:workspace',
          'fs:user-data',
          'net:http',
          'process:spawn',
          'clipboard:read',
          'clipboard:write',
          'secrets:read',
          'shell:execute',
        ],
      },
    },
    configSchema: { type: 'object' },
    disabled: { type: 'boolean' },
    contributes: {
      type: 'object',
      properties: {
        services: { type: 'array' },
        tools: { type: 'array' },
        commands: { type: 'array' },
        // ... 其他贡献点
      },
    },
  },
}

export class ManifestValidator {
  private ajv = new Ajv()
  private validate = this.ajv.compile(MANIFEST_SCHEMA)

  /**
   * 校验 manifest 结构。
   * 
   * @throws ManifestValidationError 如果校验失败
   */
  validateStructure(data: any): LumiiPluginManifest {
    const valid = this.validate(data)
    if (!valid) {
      throw new ManifestValidationError(
        `Invalid manifest: ${JSON.stringify(this.validate.errors)}`,
      )
    }
    return data as LumiiPluginManifest
  }

  /**
   * 校验 manifest 的业务规则。
   * 
   * @throws ManifestValidationError 如果校验失败
   */
  validateSemantic(manifest: LumiiPluginManifest, appVersion: string): void {
    // 检查 kind 与 main/client 的一致性
    if (manifest.kind === 'code') {
      if (!manifest.main && !manifest.client) {
        throw new ManifestValidationError(
          'kind=code requires at least main or client',
        )
      }
    }

    // 检查版本兼容性
    if (!this.semverSatisfies(appVersion, manifest.engines.lumii)) {
      throw new ManifestValidationError(
        `App version ${appVersion} does not satisfy engines.lumii: ${manifest.engines.lumii}`,
      )
    }

    // 检查权限合法性
    const validPermissions = new Set([
      'fs:workspace',
      'fs:user-data',
      'net:http',
      'process:spawn',
      'clipboard:read',
      'clipboard:write',
      'secrets:read',
      'shell:execute',
    ])
    for (const perm of manifest.permissions || []) {
      if (!validPermissions.has(perm)) {
        throw new ManifestValidationError(`Invalid permission: ${perm}`)
      }
    }
  }

  private semverSatisfies(version: string, range: string): boolean {
    // 使用 semver 库实现
    // 这里简化示例
    return true
  }
}

export class ManifestValidationError extends Error {
  name = 'ManifestValidationError'

  constructor(message: string) {
    super(message)
  }
}
```
