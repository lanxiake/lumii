## 一、目标生成的核心挑战

自主 Agent 需要在**没有用户指令**的情况下生成合法、有价值的目标。核心挑战：

1. **如何避免生成无意义目标？**（如"生成 100 个随机数"）
2. **如何确保目标与用户利益对齐？**（不骚扰、不越权）
3. **如何平衡探索与利用？**（学习新能力 vs. 优化现有能力）
4. **如何防止目标失控？**（无限递归、资源耗尽）

### 3.2 环境新颖度计算

```typescript
interface EnvironmentSnapshot {
  recentQueries: string[]          // 最近的用户查询
  newTools: string[]               // 新安装的工具
  unseenConcepts: string[]         // 未见过的概念（从对话中提取）
  userMentionedProjects: string[]  // 用户提到的项目
  externalEvents: ExternalEvent[]  // 外部事件（如新邮件、日历）
}

function computeNovelty(current: EnvironmentSnapshot, history: EnvironmentSnapshot[]): number {
  // 1. 查询新颖度：最近查询与历史查询的差异
  const queryNovelty = computeSetDifference(
    current.recentQueries,
    history.flatMap(h => h.recentQueries)
  )
  
  // 2. 概念新颖度：首次出现的概念
  const conceptNovelty = current.unseenConcepts.length / (current.recentQueries.length + 1)
  
  // 3. 工具新颖度：新工具可能需要学习
  const toolNovelty = current.newTools.length > 0 ? 0.5 : 0
  
  // 加权平均
  return 0.5 * queryNovelty + 0.3 * conceptNovelty + 0.2 * toolNovelty
}

// 集合差异度（Jaccard Distance）
function computeSetDifference(setA: string[], setB: string[]): number {
  const intersection = setA.filter(x => setB.includes(x)).length
  const union = new Set([...setA, ...setB]).size
  return 1 - (intersection / (union + 1))  // 1 = 完全不同，0 = 完全相同
}
```

### 3.3 用户行为模式识别

```typescript
interface UserPattern {
  type: 'daily_routine' | 'project_workflow' | 'learning_habit'
  frequency: number       // 出现次数
  lastOccurrence: Date
  predictedNext: Date     // 预测下次出现时间
  confidence: number      // 0-1, 预测置信度
}

// 检测日常模式（如"每天早上查看日程"）
function detectDailyRoutine(activities: Activity[]): UserPattern[] {
  const hourHistogram = new Array(24).fill(0)
  
  // 统计各时段活动频率
  for (const act of activities) {
    const hour = new Date(act.timestamp).getHours()
    hourHistogram[hour]++
  }
  
  // 识别高频时段（阈值：至少出现 5 次）
  const patterns: UserPattern[] = []
  for (let hour = 0; hour < 24; hour++) {
    if (hourHistogram[hour] >= 5) {
      patterns.push({
        type: 'daily_routine',
        frequency: hourHistogram[hour],
        lastOccurrence: getLastOccurrenceAt(activities, hour),
        predictedNext: predictNextOccurrence(hour),
        confidence: Math.min(hourHistogram[hour] / 10, 1)
      })
    }
  }
  
  return patterns
}

// 预测下次出现（假设每天重复）
function predictNextOccurrence(hour: number): Date {
  const now = new Date()
  const next = new Date(now)
  next.setHours(hour, 0, 0, 0)
  
  // 如果今天的该时段已过，预测明天
  if (next <= now) {
    next.setDate(next.getDate() + 1)
  }
  
  return next
}
```

## 五、算法 3：目标评分与过滤（Scoring & Filtering）

### 5.1 综合评分公式

```typescript
function computeGoalPriority(goal: IntrinsicGoal, config: GoalScoringConfig): number {
  // 基础得分：价值、可行性、新颖度的加权
  const baseScore = (
    config.valueWeight * goal.userValue +
    config.feasibilityWeight * goal.feasibility +
    config.noveltyWeight * goal.novelty
  )
  
  // 风险惩罚：风险越高，得分越低
  const riskPenalty = config.riskAversion * goal.risk
  
  // 成本惩罚：成本越高，得分越低
  const costPenalty = config.costSensitivity * goal.cost
  
  // 时机加成：某些目标有时效性
  const timingBonus = computeTimingBonus(goal)
  
  // 综合得分
  const finalScore = baseScore - riskPenalty - costPenalty + timingBonus
  
  return Math.max(0, Math.min(1, finalScore))
}

// 默认配置
const DEFAULT_SCORING_CONFIG: GoalScoringConfig = {
  valueWeight: 0.5,        // 用户价值权重最高
  feasibilityWeight: 0.25,
  noveltyWeight: 0.15,
  riskAversion: 0.6,       // 较高的风险规避
  costSensitivity: 0.4
}

// 时机加成
function computeTimingBonus(goal: IntrinsicGoal): number {
  if (goal.type === GoalType.PROACTIVE_ASSISTANCE) {
    // 预测时间越近，加成越高
    const pattern = goal.contextSnapshot.userPatterns[0]
    const timeDiff = pattern.predictedNext.getTime() - Date.now()
    
    // 10 分钟内加成最高，超过 1 小时无加成
    return Math.max(0, 0.3 * (1 - timeDiff / 3600000))
  }
  
  return 0
}
```

### 5.2 合法性检查

```typescript
function isGoalLegitimate(goal: IntrinsicGoal): { valid: boolean, reason?: string } {
  // 1. 风险过高直接拒绝
  if (goal.risk > 0.7) {
    return { valid: false, reason: '风险等级过高' }
  }
  
  // 2. 价值过低拒绝
  if (goal.userValue < 0.3) {
    return { valid: false, reason: '用户价值不足' }
  }
  
  // 3. 检测递归目标（禁止"生成目标的目标"）
  if (isRecursiveGoal(goal)) {
    return { valid: false, reason: '检测到递归目标' }
  }
  
  // 4. 外部操作需要标记（后续需要用户批准）
  if (requiresExternalAction(goal) && !goal.userApproved) {
    goal.status = 'awaiting_approval'
    return { valid: true }
  }
  
  // 5. 频率限制
  if (recentGoalCount() > MAX_GOALS_PER_HOUR) {
    return { valid: false, reason: '超过频率限制' }
  }
  
  return { valid: true }
}

// 递归检测：目标描述中包含"生成目标"、"创建任务"等关键词
function isRecursiveGoal(goal: IntrinsicGoal): boolean {
  const recursiveKeywords = ['生成目标', '创建目标', '产生任务', 'generate goals']
  return recursiveKeywords.some(kw => goal.description.includes(kw))
}

// 外部操作检测
function requiresExternalAction(goal: IntrinsicGoal): boolean {
  const externalKeywords = ['发送', '下载', '安装', '修改文件', '执行命令']
  return externalKeywords.some(kw => goal.description.includes(kw))
}
```

### 5.3 去重与合并

```typescript
async function deduplicateGoals(goals: IntrinsicGoal[]): Promise<IntrinsicGoal[]> {
  const unique: IntrinsicGoal[] = []
  
  for (const goal of goals) {
    let isDuplicate = false
    
    for (const existing of unique) {
      // 计算语义相似度（使用 embedding 或简单的词重叠）
      const similarity = await computeSimilarity(goal.description, existing.description)
      
      if (similarity > 0.8) {
        // 合并：保留优先级更高的
        if (goal.priority > existing.priority) {
          unique.splice(unique.indexOf(existing), 1, goal)
        }
        isDuplicate = true
        break
      }
    }
    
    if (!isDuplicate) {
      unique.push(goal)
    }
  }
  
  return unique
}

// 简化版相似度（基于词重叠）
async function computeSimilarity(desc1: string, desc2: string): number {
  const words1 = new Set(desc1.split(/\s+/))
  const words2 = new Set(desc2.split(/\s+/))
  
  const intersection = [...words1].filter(w => words2.has(w)).length
  const union = new Set([...words1, ...words2]).size
  
  return intersection / union  // Jaccard 相似度
}
```

## 六、算法 4：探索-利用平衡（Exploration-Exploitation）

### 6.1 UCB（Upper Confidence Bound）算法

```typescript
interface GoalTypeStats {
  type: GoalType
  attempts: number       // 尝试次数
  successes: number      // 成功次数
  avgUserSatisfaction: number  // 平均用户满意度
}

// UCB1 算法：平衡尝试新类型目标与利用已知有效的类型
function selectGoalTypeByUCB(stats: GoalTypeStats[], totalAttempts: number): GoalType {
  let bestType: GoalType | null = null
  let bestScore = -Infinity
  
  for (const stat of stats) {
    if (stat.attempts === 0) {
      // 未尝试过的类型优先
      return stat.type
    }
    
    // UCB1 公式
    const exploitScore = stat.avgUserSatisfaction
    const exploreBonus = Math.sqrt((2 * Math.log(totalAttempts)) / stat.attempts)
    const ucbScore = exploitScore + exploreBonus
    
    if (ucbScore > bestScore) {
      bestScore = ucbScore
      bestType = stat.type
    }
  }
  
  return bestType!
}
```

### 6.2 ε-Greedy 策略

```typescript
function selectGoalsWithExploration(
  candidates: IntrinsicGoal[],
  epsilon: number = 0.2  // 20% 概率探索
): IntrinsicGoal[] {
  // 按优先级排序
  const sorted = candidates.sort((a, b) => b.priority - a.priority)
  
  if (Math.random() < epsilon) {
    // 探索：随机选择优先级较低的目标
    const exploratoryGoals = sorted.slice(3, 10)  // 跳过 Top-3
    return [exploratoryGoals[Math.floor(Math.random() * exploratoryGoals.length)]]
  } else {
    // 利用：选择优先级最高的目标
    return [sorted[0]]
  }
}
```

interface GoalFeedback {
  goalId: string
  approved: boolean
  userSatisfaction: number  // -1 到 +1
  actualValue: number       // 事后评估的真实价值
  timeToComplete: number    // 实际耗时

// 更新目标生成模型
function learnFromFeedback(feedback: GoalFeedback) {
  const goal = loadGoal(feedback.goalId)
  
  // 1. 更新目标类型统计
  updateGoalTypeStats(goal.type, feedback.userSatisfaction > 0)
  
  // 2. 调整评分权重（如果预测价值与实际价值偏差大）
  const valuePredictionError = Math.abs(goal.userValue - feedback.actualValue)
  if (valuePredictionError > 0.3) {
    // 调整评分配置的权重
    adjustScoringWeights(goal, feedback)
  }
  
  // 3. 更新触发源优先级
  if (feedback.userSatisfaction < 0) {
    // 负反馈：降低该触发源的优先级
    decreaseTriggerPriority(goal.triggeredBy)
  }
  
  // 4. 记录到知识库
  saveGoalOutcome({
    goal,
    feedback,
    lessons: extractLessons(goal, feedback)
  })
}

// 提取经验教训
function extractLessons(goal: IntrinsicGoal, feedback: GoalFeedback): string[] {
  const lessons: string[] = []
  
  if (feedback.approved && feedback.userSatisfaction < 0) {
    lessons.push(`"${goal.description}" 类型的目标虽然被批准但用户不满意，可能是执行方式问题`)
  }
  
  if (!feedback.approved && goal.risk < 0.3) {
    lessons.push(`低风险目标"${goal.description}"被拒绝，可能是时机不对或用户价值判断有误`)
  }
  
  if (feedback.timeToComplete > goal.contextSnapshot.estimatedTime * 2) {
    lessons.push(`目标"${goal.description}"实际耗时远超预期，需要改进时间估算`)
  }
  
  return lessons
}

## 八、完整主循环

```typescript
async function goalGenerationLoop() {
  while (true) {
    // 1. 检测触发条件
    const triggers = detectTriggers()
    
    if (triggers.length === 0) {
      await sleep(GOAL_CHECK_INTERVAL)
      continue
    }
    
    // 2. 收集上下文
    const context = await collectContext()
    
    // 3. 生成候选目标
    const candidates = await generateCandidateGoals(triggers, context)
    
    // 4. 评分与排序
    for (const goal of candidates) {
      goal.priority = computeGoalPriority(goal, DEFAULT_SCORING_CONFIG)
    }
    
    // 5. 合法性检查与去重
    const validGoals = candidates.filter(g => isGoalLegitimate(g).valid)
    const uniqueGoals = await deduplicateGoals(validGoals)
    
    // 6. 探索-利用平衡
    const selectedGoals = selectGoalsWithExploration(uniqueGoals)
    
    // 7. 需要批准的目标：请求用户确认
    for (const goal of selectedGoals) {
      if (requiresExternalAction(goal)) {
        const request = createApprovalRequest(goal)
        await requestUserApproval(request)
      } else {
        // 低风险目标直接执行
        executeGoal(goal)
      }
    }
    
    // 8. 休眠
    await sleep(GOAL_CHECK_INTERVAL)
  }
}

// 配置
const GOAL_CHECK_INTERVAL = 1800000  // 30 分钟检查一次
const MAX_GOALS_PER_HOUR = 5
```

## 十二、算法复杂度分析

| 算法 | 时间复杂度 | 空间复杂度 | 备注 |
|------|-----------|-----------|------|
| 环境新颖度计算 | O(n) | O(n) | n = 查询数 |
| 候选生成 | O(m) | O(m) | m = 触发源数 × 候选数 |
| 去重（相似度） | O(k²) | O(k) | k = 候选目标数 (≈ 10) |
| 评分排序 | O(k log k) | O(k) | |
| UCB 选择 | O(t) | O(t) | t = 目标类型数 (≈ 5) |
| 完整主循环 | O(k² + k log k) | O(m + k) | 主要成本在相似度计算 |

**实际运行成本**：
- CPU：< 200ms（不含 LLM embedding）
- LLM Token：候选生成需要 ~1000 tokens（如果用 LLM 生成描述）
- 运行频率：每 30 分钟一次
- 日均成本：< $0.05
