## 一、元认知的核心问题

元认知引擎需要回答三个问题：

1. **我做得怎么样？**（Performance Assessment）
2. **我能做什么、不能做什么？**（Capability Boundary）
3. **我需要学什么？**（Learning Priority）

## 二、算法 1：满意度评分（Satisfaction Scoring）

### 2.1 目标

量化 Agent 对自身近期表现的满意度，作为触发学习的信号。

### 2.2 输入数据

```typescript
interface PerformanceMetrics {
  // 时间窗口：最近 N 轮对话
  windowSize: number
  
  // 任务完成率
  tasksCompleted: number
  tasksFailed: number
  tasksAborted: number
  
  // 用户反馈
  positiveSignals: number  // 点赞、"谢谢"、继续使用
  negativeSignals: number  // 点踩、中断、重试
  neutralSignals: number   // 无明确反馈
  
  // 工具使用效率
  toolSuccessRate: number  // 0-1
  avgToolRetries: number   // 平均重试次数
  
  // 响应质量
  avgResponseTime: number  // 毫秒
  contextOverflows: number // 上下文溢出次数
  
  // 知识覆盖
  unknownQueries: number   // "我不知道"的次数
  fallbackUsage: number    // 使用通用回答的次数
}
```

### 2.3 评分公式

```typescript
function computeSatisfactionScore(metrics: PerformanceMetrics): number {
  // 1. 任务完成维度（权重 0.35）
  const taskScore = metrics.tasksCompleted / 
    (metrics.tasksCompleted + metrics.tasksFailed + metrics.tasksAborted + 1)
  
  // 2. 用户反馈维度（权重 0.30）
  const totalFeedback = metrics.positiveSignals + metrics.negativeSignals + 1
  const feedbackScore = (
    metrics.positiveSignals - 0.5 * metrics.negativeSignals
  ) / totalFeedback
  
  // 3. 效率维度（权重 0.20）
  const efficiencyScore = (
    metrics.toolSuccessRate * 0.6 +
    (1 - Math.min(metrics.avgToolRetries / 3, 1)) * 0.4
  )
  
  // 4. 知识覆盖维度（权重 0.15）
  const knowledgeScore = 1 - (
    (metrics.unknownQueries + metrics.fallbackUsage) / 
    (metrics.windowSize + 1)
  )
  
  // 加权求和
  const rawScore = (
    0.35 * taskScore +
    0.30 * Math.max(0, feedbackScore) +  // 负反馈不低于 0
    0.20 * efficiencyScore +
    0.15 * knowledgeScore
  )
  
  // 归一化到 [0, 1]
  return Math.max(0, Math.min(1, rawScore))
}
```

### 2.4 阈值与触发逻辑

```typescript
const SATISFACTION_THRESHOLDS = {

  } else if (score >= SATISFACTION_THRESHOLDS.good) {
    return 'low'   // 探索性优化
  } else if (score >= SATISFACTION_THRESHOLDS.acceptable) {
    return 'medium'  // 主动学习
  } else {
    return 'high'  // 紧急干预
  }
}
```

```typescript
  }
```

### 3.3 边界检测算法

```typescript
// 使用 Elo Rating 系统（类似国际象棋评级）
class CapabilityRatingSystem {
  private K = 32  // 学习率
  
  updateRating(
    dimension: CapabilityDimension,
    test: CapabilityTest
  ): number {
    // 预期表现概率（基于当前能力和任务难度）
    const expected = this.expectedPerformance(
      dimension.level,
      test.difficulty
    )
    
    // 实际表现得分
    const actual = test.result === 'success' ? 1.0 :
                   test.result === 'partial' ? 0.5 : 0.0
    
    // Elo 更新公式
    const newLevel = dimension.level + this.K * (actual - expected)
    
    // 边界约束
    return Math.max(0, Math.min(1, newLevel))
  }
  
  private expectedPerformance(level: number, difficulty: number): number {
    // Logistic 函数
    const diff = level - difficulty
    return 1 / (1 + Math.exp(-10 * diff))
  }
  
  // 检测边界：在什么难度下成功率约 50%
  findBoundary(dimension: CapabilityDimension): number {
    // 求解 expectedPerformance(level, boundary) = 0.5
    return dimension.level
  }
  
  // 置信度计算：基于测试样本量
  computeConfidence(tests: CapabilityTest[]): number {
    const n = tests.length
    // 置信度随样本量增加而饱和
    return 1 - Math.exp(-n / 20)
  }
}
```

### 3.4 能力缺口识别

```typescript
interface CapabilityGap {
  dimension: string
  currentLevel: number
  desiredLevel: number
  gap: number
  priority: number  // 基于用户需求频率
}

function identifyGaps(
  capabilities: CapabilityDimension[],
  userDemand: Map<string, number>  // 维度 -> 需求频率
): CapabilityGap[] {
  const gaps: CapabilityGap[] = []
  
  for (const cap of capabilities) {
    const demand = userDemand.get(cap.name) || 0
    const desiredLevel = Math.min(0.9, 0.5 + demand * 0.5)  // 需求越高，期望水平越高
    
    if (cap.level < desiredLevel) {
      gaps.push({
        dimension: cap.name,
        currentLevel: cap.level,
        desiredLevel,
        gap: desiredLevel - cap.level,
        priority: demand * (desiredLevel - cap.level)  // 需求 × 缺口
      })
    }
  }
  
  // 按优先级降序排序
  return gaps.sort((a, b) => b.priority - a.priority)
}
```

## 四、算法 3：学习目标优先级排序

### 4.1 目标

从多个可能的学习目标中选择最应该优先处理的。

### 4.2 优先级因素

```typescript
interface LearningGoal {
  id: string
  type: 'skill' | 'knowledge' | 'prompt' | 'memory'
  description: string
  
  // 影响因素
  userValueScore: number     // 0-1, 对用户价值
  feasibility: number        // 0-1, 实现可行性
  urgency: number            // 0-1, 紧急程度
  cost: number               // 0-1, 学习成本（0=低成本）
  dependency: string[]       // 依赖的其他目标
}
```

### 4.3 优先级计算

```typescript
function computePriority(goal: LearningGoal): number {
  // 基础得分（加权平均）
  const baseScore = (
    0.40 * goal.userValueScore +
    0.25 * goal.feasibility +
    0.20 * goal.urgency +
    0.15 * (1 - goal.cost)  // 成本越低越好
  )
  
  // 依赖惩罚：有未完成依赖降低优先级
  const unmetDeps = goal.dependency.filter(dep => !isCompleted(dep)).length
  const depPenalty = Math.exp(-unmetDeps)  // e^(-n)
  
  return baseScore * depPenalty
}

// 依赖拓扑排序
function sortByDependency(goals: LearningGoal[]): LearningGoal[] {
  const sorted: LearningGoal[] = []
  const visited = new Set<string>()
  
  function dfs(goal: LearningGoal) {
    if (visited.has(goal.id)) return
    visited.add(goal.id)
    
    // 先处理依赖
    for (const depId of goal.dependency) {
      const dep = goals.find(g => g.id === depId)
      if (dep) dfs(dep)
    }
    
    sorted.push(goal)
  }
  
  // 按优先级从高到低遍历
  const prioritized = goals.sort((a, b) => 
    computePriority(b) - computePriority(a)
  )
  
  for (const goal of prioritized) {
    dfs(goal)
  }
  
  return sorted
}
```

### 4.4 学习预算分配

```typescript
interface LearningBudget {
  maxGoalsPerWeek: number
  maxTokensPerDay: number
  maxComputeTimePerDay: number  // 秒
}

function allocateBudget(
  goals: LearningGoal[],
  budget: LearningBudget
): LearningGoal[] {
  const selected: LearningGoal[] = []
  let usedTokens = 0
  let usedTime = 0
  
  for (const goal of goals) {
    const estimatedTokens = estimateTokenCost(goal)
    const estimatedTime = estimateComputeTime(goal)
    
    if (
      selected.length < budget.maxGoalsPerWeek &&
      usedTokens + estimatedTokens <= budget.maxTokensPerDay &&
      usedTime + estimatedTime <= budget.maxComputeTimePerDay
    ) {
      selected.push(goal)
      usedTokens += estimatedTokens
      usedTime += estimatedTime
    } else {
      break  // 预算耗尽
    }
  }
  
  return selected
}

// 成本估算（基于历史数据）
function estimateTokenCost(goal: LearningGoal): number {
  const baseCosts = {
    skill: 5000,      // 技能生成需要较多 token
    knowledge: 2000,  // 知识学习
    prompt: 1000,     // Prompt 优化
    memory: 500       // 记忆策略调整
  }
  return baseCosts[goal.type] * (1 + goal.cost)
}
```

### 5.3 反思结果处理

```typescript
async function conductReflection(
  trigger: ReflectionTrigger,
  metrics: PerformanceMetrics,
  capabilities: CapabilityDimension[]
): Promise<ReflectionOutput> {
  // 1. 准备上下文
  const context = {
    taskCompletionRate: metrics.tasksCompleted / 
      (metrics.tasksCompleted + metrics.tasksFailed + 1),
    userSatisfaction: computeSatisfactionScore(metrics),
    topFailures: analyzeFailures(metrics),
    strengths: capabilities.filter(c => c.level > 0.8).map(c => c.name),
    weaknesses: capabilities.filter(c => c.level < 0.4).map(c => c.name),
    gaps: identifyGaps(capabilities, getUserDemand())
  }
  
  // 2. 调用 LLM 进行反思
  const prompt = fillTemplate(REFLECTION_PROMPT, context)
  const response = await llm.generate(prompt)
  
  // 3. 解析输出（假设 JSON 格式）
  const reflection: ReflectionOutput = JSON.parse(response)
  
  // 4. 记录反思结果
  await saveReflection({
    timestamp: new Date(),
    trigger,
    metrics,
    reflection
  })
  
  // 5. 生成学习目标
  const goals = reflectionToGoals(reflection)
  await addLearningGoals(goals)
  
  return reflection
}

// 将反思转换为具体学习目标
function reflectionToGoals(reflection: ReflectionOutput): LearningGoal[] {
  return reflection.improvements
    .filter(imp => imp.priority === 'high')
    .map(imp => ({
      id: generateId(),
      type: inferGoalType(imp.description),
      description: imp.description,
      userValueScore: imp.estimatedImpact,
      feasibility: 0.7,  // 默认值，后续评估
      urgency: imp.priority === 'high' ? 0.9 : 0.5,
      cost: 0.5,
      dependency: []
    }))
}
```

## 六、元认知主循环（Meta-Cognition Loop）

### 6.1 完整流程

```typescript
async function metaCognitionLoop() {
  while (true) {
    // 1. 收集性能数据
    const metrics = collectPerformanceMetrics({ days: 7 })
    
    // 2. 计算满意度得分
    const satisfaction = computeSatisfactionScore(metrics)
    const urgency = determineLearningUrgency(satisfaction)
    
    // 3. 更新能力边界
    const capabilities = updateCapabilities(metrics)
    
    // 4. 识别能力缺口
    const gaps = identifyGaps(capabilities, getUserDemand())
    
    // 5. 判断是否需要反思
    if (
      urgency === 'high' ||
      gaps.some(g => g.priority > 0.8) ||
      shouldScheduledReflect()
    ) {
      const reflection = await conductReflection(
        ReflectionTrigger.SCHEDULED,
        metrics,
        capabilities
      )
      
      // 6. 生成学习目标
      const goals = reflectionToGoals(reflection)
      const prioritized = sortByDependency(goals)
      const selected = allocateBudget(prioritized, LEARNING_BUDGET)
      
      // 7. 执行学习任务（异步，不阻塞主循环）
      executeLearningGoals(selected)
    }
    
    // 8. 休眠直到下一个检查点
    await sleep(REFLECTION_INTERVAL)
  }
}

// 配置
const REFLECTION_INTERVAL = 6 * 3600 * 1000  // 6 小时
const LEARNING_BUDGET: LearningBudget = {
  maxGoalsPerWeek: 3,
  maxTokensPerDay: 10000,
  maxComputeTimePerDay: 300  // 5 分钟
}
```

### 6.2 性能优化

```typescript
// 增量计算：避免每次都重新计算全部指标
class MetricsCache {
  private cache: Map<string, any> = new Map()
  private ttl = 3600 * 1000  // 1 小时
  
  async get<T>(key: string, compute: () => Promise<T>): Promise<T> {
    const cached = this.cache.get(key)
    if (cached && Date.now() - cached.timestamp < this.ttl) {
      return cached.value
    }
    
    const value = await compute()
    this.cache.set(key, { value, timestamp: Date.now() })
    return value
  }
  
  invalidate(key: string) {
    this.cache.delete(key)
  }
}

// 只在有新数据时更新
function shouldUpdateMetrics(lastUpdate: Date): boolean {
  const now = new Date()
  const timeSinceUpdate = now.getTime() - lastUpdate.getTime()
  const newConversations = getConversationsSince(lastUpdate).length
  
  // 至少 1 小时 + 至少 3 个新对话
  return timeSinceUpdate > 3600000 && newConversations >= 3
}
```

## 十、算法复杂度分析

| 算法 | 时间复杂度 | 空间复杂度 | 备注 |
|------|-----------|-----------|------|
| 满意度评分 | O(1) | O(1) | 直接计算 |
| 能力边界更新 | O(n) | O(n) | n = 能力维度数 (≈ 10) |
| 能力缺口识别 | O(n log n) | O(n) | 排序主导 |
| 学习目标排序 | O(m log m + e) | O(m) | m = 目标数, e = 依赖边数 |
| 反思（LLM 调用） | O(T) | O(T) | T = LLM token 数 (~2000) |
| 完整主循环 | O(n log n + T) | O(n + m) | 主要成本在 LLM 调用 |

**实际运行成本**：
- CPU：< 100ms（不含 LLM）
- LLM Token：~2000 tokens/次反思
- 运行频率：每 6 小时一次
- 日均成本：< $0.02

```typescript
    const metrics: PerformanceMetrics = {
```
