```typescript
import { Router } from 'express';
import { autonomousCoordinator } from '@lumii/agent-runtime';

const router = Router();

// 获取状态
router.get('/status', async (req, res) => {
  try {
    const status = await autonomousCoordinator.getStatus();
    res.json(status);
  } catch (error) {
    res.status(500).json({ error: error.message });
  }
});

// 获取待批准目标
router.get('/goals', async (req, res) => {
  const { status = 'pending' } = req.query;
  try {
    const goals = await autonomousCoordinator.getGoals({ status });
    res.json({ goals });
  } catch (error) {
    res.status(500).json({ error: error.message });
  }
});

// 批准目标
router.post('/goals/:id/approve', async (req, res) => {
  const { id } = req.params;
  const { note } = req.body;
  
  try {
    const result = await autonomousCoordinator.approveGoal(id, { note });
    res.json(result);
  } catch (error) {
    res.status(400).json({ error: error.message });
  }
});

// 拒绝目标
router.post('/goals/:id/reject', async (req, res) => {
  const { id } = req.params;
  const { reason, neverAskAgain } = req.body;
  
  try {
    const result = await autonomousCoordinator.rejectGoal(id, { 
      reason, 
      neverAskAgain 
    });
    res.json(result);
  } catch (error) {
    res.status(400).json({ error: error.message });
  }
});

export default router;
```

```typescript
const BASE_URL = '/api/autonomous';

export async function getAutonomousStatus(): Promise<AutonomousStatus> {
  const res = await fetch(`${BASE_URL}/status`);
  if (!res.ok) throw new Error('Failed to fetch status');
  return res.json();
}

export async function getPendingGoals(): Promise<Goal[]> {
  const res = await fetch(`${BASE_URL}/goals?status=pending`);
  if (!res.ok) throw new Error('Failed to fetch goals');
  const data = await res.json();
  return data.goals;
}

export async function approveGoal(id: string, note?: string): Promise<Goal> {
  const res = await fetch(`${BASE_URL}/goals/${id}/approve`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ note }),
  });
  if (!res.ok) throw new Error('Failed to approve goal');
  const data = await res.json();
  return data.goal;
}

export async function rejectGoal(
  id: string, 
  options?: { reason?: string; neverAskAgain?: boolean }
): Promise<void> {
  const res = await fetch(`${BASE_URL}/goals/${id}/reject`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify(options),
  });
  if (!res.ok) throw new Error('Failed to reject goal');
}
```

```typescript
import { useQuery, useMutation } from '@tanstack/react-query';
import { getAutonomousStatus, getPendingGoals, approveGoal, rejectGoal } from '@/lib/autonomous/api';
import { SatisfactionMeter } from '@/components/autonomous/SatisfactionMeter';
import { GoalCard } from '@/components/autonomous/GoalCard';

export default function AutonomousDashboard() {
  const { data: status, isLoading } = useQuery({
    queryKey: ['autonomous-status'],
    queryFn: getAutonomousStatus,
    refetchInterval: 30000, // 每 30 秒刷新
  });

  const { data: goals = [] } = useQuery({
    queryKey: ['pending-goals'],
    queryFn: getPendingGoals,
    refetchInterval: 10000,
  });

  const approveMutation = useMutation({
    mutationFn: (id: string) => approveGoal(id),
    onSuccess: () => {
      queryClient.invalidateQueries(['pending-goals']);
    },
  });

  const rejectMutation = useMutation({
    mutationFn: (id: string) => rejectGoal(id),
    onSuccess: () => {
      queryClient.invalidateQueries(['pending-goals']);
    },
  });

  if (isLoading) return <div>加载中...</div>;

  return (
    <div className="container mx-auto p-6">
      <div className="flex justify-between items-center mb-8">
        <h1 className="text-3xl font-bold">自主进化仪表板</h1>
        <span className={`badge ${status?.enabled ? 'badge-success' : 'badge-error'}`}>
          {status?.enabled ? '✓ 已启用' : '✗ 未启用'}
        </span>
      </div>

      {/* 满意度概览 */}
      <section className="mb-8">
        <h2 className="text-xl font-semibold mb-4">📊 满意度评分</h2>
        <SatisfactionMeter 
          score={status?.satisfaction.overall || 0}
          trend={status?.satisfaction.trend}
          breakdown={status?.satisfaction.breakdown}
        />
      </section>

      {/* 待批准目标 */}
      <section>
        <h2 className="text-xl font-semibold mb-4">
          🎯 待批准目标 ({goals.length})
        </h2>
        {goals.length === 0 ? (
          <p className="text-gray-500">暂无待批准目标</p>
        ) : (
          <div className="space-y-4">
            {goals.map((goal) => (
              <GoalCard
                key={goal.id}
                goal={goal}
                onApprove={() => approveMutation.mutate(goal.id)}
                onReject={() => rejectMutation.mutate(goal.id)}
              />
            ))}
          </div>
        )}
      </section>
    </div>
  );
}
```

```typescript
interface CapabilityProgressBarProps {

export function CapabilityProgressBar({
  dimension,
  level,
  confidence,
  testCount,
  trend,
}: CapabilityProgressBarProps) {
  const trendIcon = trend === 'up' ? '↑' : trend === 'down' ? '↓' : '→';
  const trendColor = trend === 'up' ? 'text-green-500' : trend === 'down' ? 'text-red-500' : 'text-gray-500';

  return (
    <div className="mb-4">
      <div className="flex justify-between items-center mb-2">
        <span className="font-medium">{dimensionLabels[dimension]}</span>
        <span className={`text-sm ${trendColor}`}>
          {(level * 100).toFixed(0)}% {trendIcon}
        </span>
      </div>
      <div className="relative h-6 bg-gray-200 rounded-full overflow-hidden">

          className="h-full bg-blue-500 transition-all duration-500"

        />
```

## 十、离线审批架构（核心修正）

### 10.1 原方案的缺陷

初版方案把审批入口放在前端仪表板 + 桌面通知，这在真实使用中不成立：

| 缺陷 | 后果 |
|------|------|
| 用户不常守在电脑前 | 目标永久停在 `pending`，进化链路彻底断裂 |
| 桌面通知需应用运行且有权限 | 通知丢失即审批丢失，无补偿 |
| 无超时策略 | `pending` 目标无限堆积，每日配额被占满后不再生成新目标 |
| 所有目标一律需批准 | 纯内部、无副作用的改进也被阻塞，进化速度趋近于零 |

**当前代码实证**：`intrinsic-goal-generator.ts:223` 生成后仅写库为 `PENDING`；
`approveGoal()`（`:238`）只是一条等人调用的 SQL。**没有任何通知动作，没有超时**。
即目标生成后事实上无人可批。

### 10.2 三层解决方案

```
目标生成
  ↓
① 风险分级 ── 无副作用 ──→ 自动批准（记审计，可回滚）
  ↓ 需人工确认
② 渠道送达 ── 飞书/微信/企微推送 ──→ 用户回复 1/2/3 ──→ 回填
  ↓ 超时未回复
③ 超时兜底 ── 按类型走「安全侧」默认动作（不是一律批准）
```

### 10.3 第一层：风险分级与自动审批

**核心判断：不是所有目标都需要人工审批。** 按副作用范围分级：

| 级别 | 目标类型 | 副作用范围 | 审批策略 |
|------|---------|-----------|---------|
| **L0 内部** | `learning`、`capability-improvement`、`memory-optimization` | 只改 Agent 自身内部状态（能力评分、记忆权重、Prompt 变体） | **自动批准**，记审计，可回滚 |
| **L1 对外可见** | `proactive-message` | 主动给用户发消息，会打扰用户 | **需审批**，并受频率限制 |
| **L2 有副作用** | `skill-enhancement`、任何涉及文件写入/命令执行/权限变更/网络写操作 | 可能改变用户环境 | **始终需人工审批**，永不自动批准 |

```typescript
export type GoalRiskLevel = 'internal' | 'user-visible' | 'side-effect'

/**
 * 目标风险分级。
 *
 * 未知类型一律降级为最严格的 side-effect，避免新增类型默认被自动放行。
 */
export function classifyGoalRisk(goal: AutonomousGoal): GoalRiskLevel {
  if (requiresExternalAction(goal)) return 'side-effect'

  switch (goal.type) {
    case GoalType.LEARNING:
    case GoalType.CAPABILITY_IMPROVEMENT:
    case GoalType.MEMORY_OPTIMIZATION:
      return 'internal'
    case GoalType.PROACTIVE_MESSAGE:
      return 'user-visible'
    case GoalType.SKILL_ENHANCEMENT:
      return 'side-effect'
    default:
      return 'side-effect' // 保守默认
  }
}

/** 关键词兜底：描述中出现外部动作即升级为 side-effect */
function requiresExternalAction(goal: AutonomousGoal): boolean {
  const patterns = [
    '发送', '下载', '安装', '删除', '修改文件', '执行命令', '推送', '提交',
  ]
  return patterns.some((p) => goal.description.includes(p))
}
```


### 10.4 第二层：渠道送达（复用现有渠道设施）

项目已有完整的多渠道出站能力，**不要新建通知体系**：

| 现有能力 | 位置 | 复用方式 |
|---------|------|---------|
| `ChannelOutboundRouter.send()` | `apps/windows/src/main/channel/channel-outbound-router.ts:37` | 推送审批请求，硬失败带 `errorCode` |
| 四个渠道适配器（飞书/微信/企微/IPC） | `apps/windows/src/main/channel/adapters/` | 无需改动 |
| cron 的 `notify_targets` 多目标派发 | `apps/windows/src/main/agent-runtime/cron-scheduler.ts:661` | 直接照搬派发模式 |
| 审批文案与回复解析 | `channel-interaction-store.ts`（`formatPermissionPrompt` / `parsePermissionReply`） | 复用文案与 1/2/3 解析 |

#### 为什么不能直接复用 `ChannelInteractionHub`

`ChannelInteractionHub` 已实现渠道审批（`channel-interaction-hub.ts:61`、`:95`），但它有两个前提与自主进化不匹配：

1. **依赖活跃会话**：`onInteraction()` 需要 `routes.get(req.sessionKey)` 命中，
   而 `routes` 只在用户有入站消息时由 `trackSession()` 填充（`:43`）。
   自主进化由 cron 后台触发，没有活跃会话 → 直接 `return false`，推送不出去。
2. **内存态 + 同步阻塞**：交互存在内存 `Map` 中，且 bridge 的 resolver 正阻塞着一轮 agent turn。
   而目标审批是**长时异步**的，可能几小时后才回复，进程重启不能丢。

**结论**：复用其**文案格式化与回复解析**，但审批状态必须**持久化到数据库**，
并走 `ChannelOutboundRouter` 主动推送（不依赖 replyContext）。

#### 审批队列表

```sql
CREATE TABLE autonomous_approvals (
  id TEXT PRIMARY KEY,
  goal_id TEXT NOT NULL,
  risk_level TEXT NOT NULL,          -- internal | user-visible | side-effect
  status TEXT NOT NULL,              -- pending | approved | rejected | expired
  -- 送达
  channel TEXT,                      -- feishu | weixin | wecom | local
  peer_id TEXT,
  delivery_status TEXT,              -- sent | failed | unreachable
  delivery_error TEXT,               -- 失败时的 errorCode
  delivered_at INTEGER,
  -- 决策
  decided_by TEXT,                   -- user | auto-policy | timeout-policy
  decided_at INTEGER,
  decision_note TEXT,
  -- 超时
  expires_at INTEGER NOT NULL,
  created_at INTEGER NOT NULL,
  FOREIGN KEY (goal_id) REFERENCES autonomous_goals(id)
);

CREATE INDEX idx_approvals_pending ON autonomous_approvals(status, expires_at);
CREATE UNIQUE INDEX idx_approvals_goal ON autonomous_approvals(goal_id);
```

`idx_approvals_goal` 唯一约束保证一个目标只有一条审批记录，重试不会重复推送。

#### 送达流程

```typescript
/**
 * 推送审批请求到渠道。
 *
 * 送达失败不抛错：标记 delivery_status 让超时策略接管，
 * 否则一次网络抖动就会让目标永久卡住。
 */
async function deliverApproval(
  approval: ApprovalRecord,
  goal: AutonomousGoal,
): Promise<void> {
  const router = getChannelRouter()
  if (!router) {
    await markUnreachable(approval.id, 'HUB_NOT_READY')
    return
  }

  // 选择可送达的 peer：优先用户配置的渠道，回退到任一 canSend 的 peer
  const target = await resolveApprovalTarget(router)
  if (!target) {
    await markUnreachable(approval.id, 'PEER_NOT_FOUND')
    return
  }

  const result = await router.send({
    channel: target.channel,
    to: target.peerId,
    text: formatGoalApprovalPrompt(goal), // 复用渠道审批文案风格
  })

  if (!result.ok) {
    await markUnreachable(approval.id, result.errorCode ?? 'UPSTREAM_ERROR')
    return
  }
  await markDelivered(approval.id, target)
}
```

**推送文案**（复用 1/2/3 语义，与现有渠道审批一致）：

```
🎯 我想做一件事，需要你确认

[能力提升] 提升代码生成能力：从 52% 提升到 70%
原因：最近 7 次代码任务中有 3 次你不满意

回复 1 = 同意  2 = 拒绝  3 = 以后同类都自动同意

（4 小时内未回复将自动跳过，不会打扰你）
```

最后一行很重要：**提前告知超时行为**，用户不回复也知道会发生什么。

#### 回复回填

用户回复由渠道 adapter 的 out-of-band 拦截路径处理（复用
`tryHandleChannelOutOfBand` 的位置，`channel-interaction-hub.ts:156`），
但匹配的是数据库中的待审批项而非内存 Map：

```typescript
/**
 * 尝试把入站文本当作目标审批回复消费。
 *
 * 只消费最近一条待审批项，避免用户一句「1」被多条审批同时吃掉。
 */
export async function tryConsumeGoalApproval(
  peerId: string,
  text: string,
): Promise<boolean> {
  const pending = await findLatestPendingApproval(peerId)
  if (!pending) return false

  const decision = parseApprovalReply(text) // 1/2/3 → approve/reject/always
  if (!decision) return false // 看不懂就不消费，交回正常对话流程

  await applyDecision(pending, decision, 'user')
  return true
}
```

**边界处理：**
- 解析不出 1/2/3 → **不消费**，当普通消息处理（避免吃掉用户正常对话）
- 多条待审批 → 只消费最近一条，并在回执中说明还剩几条
- 已过期的审批收到回复 → 明确回复「该请求已过期」，不静默丢弃

### 10.5 第三层：超时兜底（默认动作偏向安全侧）

**关键原则：超时的默认动作不是统一自动批准，而是每类目标各自的「安全侧」。**

| 目标类型 | 超时时长 | 超时动作 | 理由 |
|---------|---------|---------|------|
| `learning` | 4h | **批准** | 纯内部学习，做了无风险，不做则无进化 |
| `capability-improvement` | 4h | **批准** | 同上 |
| `memory-optimization` | 4h | **批准**（仅标记，不删除） | 只调排序权重，删除仍需显式批准 |
| `proactive-message` | 2h | **拒绝** | 过期的主动消息发出去反而是骚扰 |
| `skill-enhancement` | 24h | **过期归档** | 有副作用，超时绝不批准；归档避免占用配额 |

```typescript
interface TimeoutPolicy {
  ttlMs: number
  onTimeout: 'approve' | 'reject' | 'expire'
}

export const TIMEOUT_POLICIES: Record<GoalType, TimeoutPolicy> = {
  [GoalType.LEARNING]:                { ttlMs: 4 * 3600_000,  onTimeout: 'approve' },
  [GoalType.CAPABILITY_IMPROVEMENT]:  { ttlMs: 4 * 3600_000,  onTimeout: 'approve' },
  [GoalType.MEMORY_OPTIMIZATION]:     { ttlMs: 4 * 3600_000,  onTimeout: 'approve' },
  [GoalType.PROACTIVE_MESSAGE]:       { ttlMs: 2 * 3600_000,  onTimeout: 'reject'  },
  [GoalType.SKILL_ENHANCEMENT]:       { ttlMs: 24 * 3600_000, onTimeout: 'expire'  },
}
```

**送达失败时的处理**（重要）：

```typescript
/**
 * 送达失败（unreachable）时收紧策略：
 * 用户根本没收到请求，不能当作「默示同意」处理有对外影响的目标。
 */
function resolveTimeoutAction(approval: ApprovalRecord, goal: AutonomousGoal) {
  const policy = TIMEOUT_POLICIES[goal.type]

  // 从未送达 → 只允许 L0 内部目标按 approve 走，其余一律 expire
  if (approval.deliveryStatus !== 'sent') {
    return approval.riskLevel === 'internal' ? policy.onTimeout : 'expire'
  }
  return policy.onTimeout
}
```

**扫描任务**：复用 cron 调度（`cron-scheduler.ts`），每 15 分钟扫描一次
`status = 'pending' AND expires_at < now`，按策略结算。

### 10.6 用户可配置项

审批策略必须由用户掌控，全部可配置：

```typescript
export interface ApprovalSettings {
  /** 审批送达渠道；'local' 表示只在应用内审批 */
  channel: 'feishu' | 'weixin' | 'wecom' | 'local' | 'off'
  /** 渠道内收件人 peer id */
  peerId?: string
  /** 是否自动批准 L0 内部目标 */
  autoApproveInternal: boolean
  /** 各类型超时时长覆盖（毫秒） */
  ttlOverrides?: Partial<Record<GoalType, number>>
  /** 免打扰时段，此间不推送审批（顺延至结束） */
  quietHours?: { start: number; end: number }
}
```

免打扰时段的处理：**顺延而非丢弃**，并相应延长 `expires_at`，
否则夜间生成的目标会在用户醒来前就超时结算。

### 10.10 必测场景

1. **完全离线**：应用运行、用户 8 小时不碰电脑 → L0 自动批准，`proactive-message` 超时拒绝，`skill-enhancement` 归档
2. **渠道送达失败**：飞书未登录 → 标记 unreachable，非 L0 目标一律 expire，不按默示同意处理
3. **延迟回复**：推送后 3 小时回复「1」→ 若未超时则正常批准；已超时则明确回复已过期
4. **重启恢复**：推送后重启应用 → 待审批项与超时时间不丢
5. **免打扰**：23:30 生成目标 → 顺延到 08:00 推送，`expires_at` 相应延长
6. **回复歧义**：回复「好的」→ 不消费，走正常对话；回复「1」才消费
7. **多条待审批**：连续 3 条 → 回复「1」只结算最近一条，回执提示剩余
8. **自动决策回滚**：自动批准后用户回滚 → 目标回到 rejected，效果撤销
