### 3.3 packages/plugin-core/src/context.ts（部分）

```typescript
import { EventBus } from './event-bus'
import { PluginScope } from './scope'
import { type LumiiEvents, type ServiceKey } from './types'
import {
  DuplicateServiceError,
  MissingServiceError,
  ScopeAlreadyDisposedError,
} from './errors'

export interface PluginLogger {
  debug(message: string, data?: any): void
  info(message: string, data?: any): void
  warn(message: string, data?: any): void
  error(message: string, error?: any): void
}

class DefaultPluginLogger implements PluginLogger {
  constructor(private pluginId: string) {}

  debug(message: string, data?: any) {
    console.debug(`[${this.pluginId}] ${message}`, data)
  }

  info(message: string, data?: any) {
    console.info(`[${this.pluginId}] ${message}`, data)
  }

  warn(message: string, data?: any) {
    console.warn(`[${this.pluginId}] ${message}`, data)
  }

  error(message: string, error?: any) {
    console.error(`[${this.pluginId}] ${message}`, error)
  }
}

export interface PluginContext {
  readonly pluginId: string
  provide<T>(key: ServiceKey<T>, value: T | (() => T)): T
  get<T>(key: ServiceKey<T>): T | undefined
  require<T>(key: ServiceKey<T>): T
  effect(dispose: () => void | Promise<void>): void
  on<K extends keyof LumiiEvents>(name: K, listener: (payload: LumiiEvents[K]) => void | Promise<void>): void
  off<K extends keyof LumiiEvents>(name: K, listener: (payload: LumiiEvents[K]) => void | Promise<void>): void
  emit<K extends keyof LumiiEvents>(name: K, payload: LumiiEvents[K]): void
  waterfall<K extends keyof LumiiEvents>(name: K, payload: LumiiEvents[K]): Promise<LumiiEvents[K]>
  serial<K extends keyof LumiiEvents>(name: K, payload: LumiiEvents[K]): Promise<LumiiEvents[K]>
  parallel<K extends keyof LumiiEvents>(name: K, payload: LumiiEvents[K]): Promise<void>
  readonly log: PluginLogger
}

export function createPluginContext(
  pluginId: string,
  eventBus: EventBus,
  scope: PluginScope,
): PluginContext {
  const services = new Map<ServiceKey<any>, any>()
  const serviceFactories = new Map<ServiceKey<any>, () => any>()

  return {
    pluginId,

    provide<T>(key: ServiceKey<T>, value: T | (() => T)): T {
      if (services.has(key) || serviceFactories.has(key)) {
        throw new DuplicateServiceError(pluginId, String(key))
      }

      if (typeof value === 'function') {
        serviceFactories.set(key, value as () => T)
        scope.effect(() => {
          serviceFactories.delete(key)
        })
        return undefined as any
      } else {
        services.set(key, value)
        scope.effect(() => {
          services.delete(key)
        })
        return value
      }
    },

    get<T>(key: ServiceKey<T>): T | undefined {
      if (services.has(key)) return services.get(key)
      if (serviceFactories.has(key)) {
        const factory = serviceFactories.get(key)!
        const instance = factory()
        services.set(key, instance)
        serviceFactories.delete(key)
        return instance
      }
      return undefined
    },

    require<T>(key: ServiceKey<T>): T {
      const svc = this.get(key)
      if (svc === undefined) {
        throw new MissingServiceError(pluginId, String(key))
      }
      return svc
    },

    effect(dispose: () => void | Promise<void>): void {
      scope.effect(dispose)
    },

    on<K extends keyof LumiiEvents>(
      name: K,
      listener: (payload: LumiiEvents[K]) => void | Promise<void>,
    ): void {
      eventBus.on(name, listener)
      scope.effect(() => {
        eventBus.off(name, listener)
      })
    },

    off<K extends keyof LumiiEvents>(
      name: K,
      listener: (payload: LumiiEvents[K]) => void | Promise<void>,
    ): void {
      eventBus.off(name, listener)
    },

    emit<K extends keyof LumiiEvents>(name: K, payload: LumiiEvents[K]): void {
      eventBus.emit(name, payload)
    },

    waterfall<K extends keyof LumiiEvents>(
      name: K,
      payload: LumiiEvents[K],
    ): Promise<LumiiEvents[K]> {
      return eventBus.waterfall(name, payload)
    },

    serial<K extends keyof LumiiEvents>(
      name: K,
      payload: LumiiEvents[K],
    ): Promise<LumiiEvents[K]> {
      return eventBus.serial(name, payload)
    },

    parallel<K extends keyof LumiiEvents>(
      name: K,
      payload: LumiiEvents[K],
    ): Promise<void> {
      return eventBus.parallel(name, payload)
    },

    log: new DefaultPluginLogger(pluginId),
  }
}
```

### 4.1 packages/plugin-core/src/__tests__/scope.test.ts（示例）

```typescript
import { describe, it, expect, beforeEach } from 'vitest'
import { PluginScope } from '../scope'

describe('PluginScope', () => {
  let scope: PluginScope

  beforeEach(() => {
    scope = new PluginScope()
  })

  it('should execute disposers in LIFO order', async () => {
    const order: number[] = []

    scope.effect(() => order.push(1))
    scope.effect(() => order.push(2))
    scope.effect(() => order.push(3))

    const result = await scope.dispose()
    expect(result.ok).toBe(true)
    expect(order).toEqual([3, 2, 1])
  })

  it('should wait for async disposers serially', async () => {
    const timeline: string[] = []

    scope.effect(async () => {
      timeline.push('A-start')
      await new Promise((resolve) => setTimeout(resolve, 10))
      timeline.push('A-end')
    })

    scope.effect(async () => {
      timeline.push('B-start')
      await new Promise((resolve) => setTimeout(resolve, 5))
      timeline.push('B-end')
    })

    await scope.dispose()
    // B 先启动（LIFO），然后等 B 完成才启动 A
    expect(timeline).toEqual(['B-start', 'B-end', 'A-start', 'A-end'])
  })

  it('should continue cleanup even if one disposer fails', async () => {
    let cleaned = false

    scope.effect(() => {
      throw new Error('Disposer error')
    })

    scope.effect(() => {
      cleaned = true
    })

    const result = await scope.dispose()
    expect(result.ok).toBe(false)
    expect(result.errors).toHaveLength(1)
    expect(cleaned).toBe(true) // 继续清理
  })

  it('should be idempotent', async () => {
    let count = 0
    scope.effect(() => count++)

    const r1 = await scope.dispose()
    const r2 = await scope.dispose()

    expect(r1.ok).toBe(true)
    expect(r2.ok).toBe(true)
    expect(count).toBe(1) // 只执行一次
  })

  it('should throw ScopeAlreadyDisposedError on late effect registration', async () => {
    await scope.dispose()
    expect(() => {
      scope.effect(() => {})
    }).toThrow('ScopeAlreadyDisposedError')
  })
})
```

## 5. IPC 归属治理

> **实施时的偏离（已落地）**：原文打算给 194 个通道逐个写 `owner/schema/risk`。实际做下来
> 改成**按注册文件登记归属**——`scripts/ipc-owners.json` 25 行，由 `check-wiring.mjs` 的
> 第 ⑩ 项守卫。理由：逐通道 194 行在 P0 没有任何消费者（控制面与 PluginHost 要到 P1 才有），
> 维护成本高、退化风险大；而「这个通道是谁的」这个问题按文件回答即可完整覆盖。
> `schema` / `risk` 推迟到 P1 控制面真正需要时再补，那时由 manifest 提供。

### 5.1 归属表 `scripts/ipc-owners.json`

键是**注册 `ipcMain.handle/on` 的文件**（相对 `apps/windows/src/main/`），值是 owner 标签。

```json
{
  "ipc/window-ipc.ts": "kernel",
  "cloud-sync/sync-ipc.ts": "cloud-sync",
  "voice/voice-ipc.ts": "voice"
}
```

实测 25 个文件注册 IPC；守卫在 `check-wiring.mjs` 第 ⑩ 项：

- 注册了 IPC 但没登记 → 报**新断线**（棘轮，`--strict` 下退出 1）；
- 登记了但已不再注册 → 提示可删除（不算失败）。

通道级映射不落盘：需要时从 `handlers` 集合现场生成，避免第二份清单。

### 5.2 后续（P1 再做）

- `schema` / `risk` 由 manifest 的 `contributes.ipc` 提供；
- 控制面白名单从已激活插件聚合。

### 6.2 plugin-manifests 检查

```bash
# scripts/check-plugin-manifests.mjs (NEW)

/**
 * 检查所有 manifest 的完整性与冲突。
 * 
 * 检查内容：
 * - 每个 manifest id 唯一
 * - 每个 manifest 结构有效（通过 JSON Schema）
 * - 版本号符合 semver
 * - 内置插件 id 以 com.lumii 开头
 * - 不存在循环依赖（基于 inject/provides）
 */

import { globSync } from 'glob'
import { readFileSync } from 'fs'
import { ManifestValidator } from '@mtbot/plugin-core'

const validator = new ManifestValidator()
const manifests = new Map()

// 扫描所有 manifest 文件
const patterns = [
  'apps/windows/src/plugins/*/package.json',  // 内置插件
  'plugins/*/package.json',                    // 本地插件（如果有）
]

for (const pattern of patterns) {
  for (const file of globSync(pattern)) {
    const pkg = JSON.parse(readFileSync(file, 'utf-8'))
    const manifest = pkg.lumii || (pkg.plugin && readFileSync(pkg.plugin, 'utf-8').then(JSON.parse))

    if (manifest) {
      if (manifests.has(manifest.id)) {
        throw new Error(`Duplicate manifest id: ${manifest.id}`)
      }

      validator.validateStructure(manifest)
      validator.validateSemantic(manifest, process.env.LUMII_VERSION || '1.0.0')

      manifests.set(manifest.id, manifest)
    }
  }
}

console.log(`✓ Found ${manifests.size} manifest(s)`)
console.log(`✓ All manifests are valid`)
```

### 6.3 ipc-ownership 检查

```bash
# scripts/check-ipc-ownership.mjs (NEW)

/**
 * 检查所有 IPC 通道是否有 owner 元数据。
 * 
 * 检查内容：
 * - 211 个通道都在 IPC_METADATA 中定义
 * - 每个通道有 owner / risk / description
 * - 没有孤立的元数据（引用不存在的通道）
 * - owner 符合 'kernel' 或插件 id 格式
 */

import { readFileSync } from 'fs'

const registry = readFileSync('apps/windows/src/main/ipc/ipc-handlers-registry.ts', 'utf-8')
const metadata = readFileSync('apps/windows/src/main/ipc/ipc-metadata.ts', 'utf-8')

// 提取通道名称
const registryChannels = new Set()
const regex = /ipcMain\.handle\(['"]([^'"]+)['"]/g
let match
while ((match = regex.exec(registry)) !== null) {
  registryChannels.add(match[1])
}

// 提取元数据中定义的通道
const metadataChannels = new Set()
const metaRegex = /['"]([^'"]+)['"]\s*:\s*\{/g
while ((match = metaRegex.exec(metadata)) !== null) {
  if (match[1].includes(':')) {
    metadataChannels.add(match[1])
  }
}

// 检查
const missing = Array.from(registryChannels).filter((c) => !metadataChannels.has(c))
const orphaned = Array.from(metadataChannels).filter((c) => !registryChannels.has(c))

if (missing.length > 0) {
  throw new Error(`Missing IPC metadata for: ${missing.join(', ')}`)
}

if (orphaned.length > 0) {
  console.warn(`Orphaned IPC metadata: ${orphaned.join(', ')}`)
}

console.log(`✓ ${registryChannels.size} IPC channel(s) have ownership metadata`)
```

## 9. 风险与回滚

### 9.1 关键风险点

| 风险 | 症状 | 缓解 |
|---|---|---|
| plugin-core 类型复杂 | serviceKey 推导失败 | 早期进行 typecheck；单测逐个验证 |
| manifest schema 过严 | 现有配置无法加载 | 先校验现有 manifest；确保向后兼容 |
| IPC 元数据重复/冲突 | 启动异常或白屏 | 自动检查脚本；手工 code review |
| bootstrap 分阶段引入顺序问题 | App 启动顺序变化 | 原 590 行顺序完全保留；只是分组 |

### 9.2 回滚触发条件

任何一个满足立即回滚：
- ❌ `pnpm test:all` 失败（fail > 0）
- ❌ `pnpm typecheck` 新增错误
- ❌ 启动耗时 > ±100ms（相比基线）
- ❌ 任何已验证功能不工作

### 9.3 回滚步骤

```bash
# 如果需要全面回滚
git revert <P0-commit-range>

# 或者逐个回滚
git revert <specific-commit>

# 恢复基线
pnpm install
pnpm verify
# 重启 App，验证功能恢复
```
